#!/usr/bin/env python3 import subprocess, sys from pathlib import Path def process(url, model="htdemucs"): out = Path("stems") out.mkdir(exist_ok=True) print("Downloading...") subprocess.run([ "yt-dlp", "-x", "--audio-format", "wav", "--audio-quality", "0", "-o", str(out / "%(id)s.%(ext)s"), "--no-playlist", url ], check=True) audio = next(out.glob("*.wav")) print(f"Separating with {model}...") subprocess.run(["demucs", "-n", model, "--out", str(out), str(audio)], check=True) audio.unlink() print(f"\nStems ready in: {out / model / audio.stem}") print("Drop those WAV files into the Voice DAW.") if __name__ == "__main__": if len(sys.argv) < 2: print("Usage: python get_stems.py ") sys.exit(1) process(sys.argv[1])
